# Monet - Execution Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build an OS where you type what you want, agents do the work, and the system shows results in the optimal UI pattern for that task. You approve or reject. Done.

**Architecture:** Debian (stripped) boots into Sway, which launches a fullscreen Flutter shell. The shell talks to a Python/FastAPI agent backend over Unix socket. Agents use the Claude Agent SDK with Nango for tool integrations. The system classifies user intent, picks the right agent and UI pattern, and pauses at approval gates before executing actions.

**Tech Stack:**

| Layer           | Choice                          | Why                                                                               |
| --------------- | ------------------------------- | --------------------------------------------------------------------------------- |
| Base            | Debian 12 aarch64 (stripped)    | Stable, massive ecosystem, proven forkable                                        |
| Compositor      | Sway                            | Off-the-shelf Wayland compositor, don't build what we don't need                  |
| Shell UI        | Flutter (native Wayland client) | Compiles native, gestures (swipe), canvas (whiteboard), hot reload, future mobile |
| Agent backend   | Python + FastAPI                | Claude Agent SDK is Python-first. Runs as systemd service                         |
| Agent framework | Claude Agent SDK                | Production loop, approval hooks, MCP, subagents, streaming                        |
| Integrations    | Nango                           | Unified API for 250+ integrations, managed OAuth, webhooks, sync                  |
| State           | SQLite                          | Local, simple, no server                                                          |
| IPC             | Unix socket / HTTP localhost    | Shell <-> agent backend                                                           |

```python
    metadata: dict = field(default_factory=dict)

@dataclass
class AgentResult:
    agent: str
    ui_pattern: str
    outputs: list[AgentOutput] = field(default_factory=list)

    type: str  # "token" | "tool_call" | "approval_request" | "done"
    data: str = ""

        self.sessions = {}

        events = list(self.stream_sync(intent, session_id))
        # Collect final result from events
        outputs = []
        agent_name = "general"
        ui_pattern = "chat"

            if event.type == "done":
                agent_name = event.metadata.get("agent", agent_name)
                ui_pattern = event.metadata.get("ui_pattern", ui_pattern)
            if event.type == "token":
                outputs.append(AgentOutput(content=event.data))
        return AgentResult(agent=agent_name, ui_pattern=ui_pattern, outputs=outputs)
```

```python
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from .runner import AgentRunner
import json

app = FastAPI(title="Monet Agent Backend")
runner = AgentRunner()

class IntentRequest(BaseModel):
    intent: str
    session_id: str = None

@app.post("/api/run")
async def run_agent(req: IntentRequest):
    result = runner.run_sync(req.intent, req.session_id)
    return {
        "agent": result.agent,
        "ui_pattern": result.ui_pattern,
        "outputs": [
            {"content": o.content, "status": o.status, "metadata": o.metadata}
            for o in result.outputs
        ],
    }

@app.post("/api/stream")
async def stream_agent(req: IntentRequest):
    def generate():
        for event in runner.stream_sync(req.intent, req.session_id):
            yield json.dumps({
                "type": event.type,
                "data": event.data,
                "metadata": event.metadata,
            }) + "\n"
    return StreamingResponse(generate(), media_type="application/x-ndjson")

@app.get("/api/health")
async def health():
    return {"status": "ok"}
```

```python
# agent/router.py
from dataclasses import dataclass
import re

@dataclass
class RoutedIntent:
    agent: str
    ui_pattern: str  # "tinder" | "chat" | "diff" | "whiteboard"
    original: str

# Keyword rules: (pattern, agent, ui_pattern)
RULES = [
    # Email - batch
    (r"\b(inbox|handle.*emails?|triage|batch.*email)\b", "email", "tinder"),
    # Email - single
    (r"\b(reply|respond|draft|write.*email|send.*email|forward|follow.?up)\b", "email", "chat"),
    # Code - review
    (r"\b(review|pr|pull.?request|diff|code.?review)\b", "code", "diff"),
    # Code - generate
    (r"\b(write.*code|build|implement|fix.*bug|generate.*code)\b", "code", "chat"),
    # Planning
    (r"\b(plan|sprint|brainstorm|roadmap|ideas|organize|prioritize)\b", "planning", "whiteboard"),
]

class IntentRouter:
    def route(self, intent: str) -> RoutedIntent:
        text = intent.lower()
        for pattern, agent, ui_pattern in RULES:
            if re.search(pattern, text):
                return RoutedIntent(agent=agent, ui_pattern=ui_pattern, original=intent)

        # Default: general agent, chat UI
        return RoutedIntent(agent="general", ui_pattern="chat", original=intent)
```

```python
# agent/approval.py
from dataclasses import dataclass, field
from enum import Enum
import uuid

class ApprovalStatus(str, Enum):
    PENDING = "pending"
    APPROVED = "approved"
    REJECTED = "rejected"

@dataclass
class ApprovalRequest:
    id: str
    tool_name: str
    parameters: dict
    status: ApprovalStatus = ApprovalStatus.PENDING

class ApprovalGate:
    def __init__(self):
        self.requests: dict[str, ApprovalRequest] = {}

    def create(self, tool_name: str, parameters: dict) -> ApprovalRequest:
        req = ApprovalRequest(
            id=str(uuid.uuid4())[:8],
            tool_name=tool_name,
            parameters=parameters,
        )
        self.requests[req.id] = req
        return req

    def get(self, request_id: str) -> ApprovalRequest:
        return self.requests[request_id]

    def approve(self, request_id: str):
        self.requests[request_id].status = ApprovalStatus.APPROVED

    def reject(self, request_id: str):
        self.requests[request_id].status = ApprovalStatus.REJECTED

    def list_pending(self) -> list[ApprovalRequest]:
        return [r for r in self.requests.values() if r.status == ApprovalStatus.PENDING]
```

```python
# agent/cli.py
"""
CLI test harness for Monet agents.
Usage: python -m agent.cli "handle my inbox"
"""
import sys
import asyncio
from .runner import AgentRunner
from .approval import ApprovalGate

runner = AgentRunner()
gate = ApprovalGate()

async def main():
    if len(sys.argv) < 2:
        print("Usage: python -m agent.cli <intent>")
        print('Example: python -m agent.cli "handle my inbox"')
        sys.exit(1)

    intent = " ".join(sys.argv[1:])
    print(f"\n> {intent}\n")

    # Stream events
    for event in runner.stream_sync(intent):
        if event.type == "token":
            print(event.data, end="", flush=True)
        elif event.type == "approval_request":
            req = gate.create(
                event.metadata["tool_name"],
                event.metadata["parameters"],
            )
            print(f"\n\n[APPROVAL REQUIRED] {req.tool_name}")
            print(f"  Parameters: {req.parameters}")
            response = input("  Approve? (y/n): ").strip().lower()
            if response == "y":
                gate.approve(req.id)
                print("  -> Approved")
            else:
                gate.reject(req.id)
                print("  -> Rejected")
        elif event.type == "done":
            print(f"\n\nAgent: {event.metadata.get('agent')}")
            print(f"UI Pattern: {event.metadata.get('ui_pattern')}")

if __name__ == "__main__":
    asyncio.run(main())
```

```dart
    final resp = await http.post(
      Uri.parse('$baseUrl/api/run'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({'intent': intent}),
    );
    return jsonDecode(resp.body);
  }

  Stream<Map<String, dynamic>> stream(String intent) async* {
    final client = http.Client();
    final request = http.Request('POST', Uri.parse('$baseUrl/api/stream'));
    request.headers['Content-Type'] = 'application/json';
    request.body = jsonEncode({'intent': intent});

    final response = await client.send(request);
    await for (final chunk in response.stream.transform(utf8.decoder)) {
      for (final line in chunk.split('\n')) {
        if (line.trim().isNotEmpty) {
          yield jsonDecode(line);
        }
      }
    }
    client.close();
  }

  Future<List<Map<String, dynamic>>> getApprovals() async {
    final resp = await http.get(Uri.parse('$baseUrl/api/approvals'));
    return List<Map<String, dynamic>>.from(jsonDecode(resp.body));
  }

  Future<void> approve(String id) async {
    await http.post(Uri.parse('$baseUrl/api/approvals/$id/approve'));
  }

  Future<void> reject(String id) async {
    await http.post(Uri.parse('$baseUrl/api/approvals/$id/reject'));
  }
}
```

```dart
import 'package:flutter/material.dart';

class TinderPattern extends StatefulWidget {
  final List<Map<String, dynamic>> items;
  final Function(int index, bool approved) onDecision;

  const TinderPattern({
    super.key,
    required this.items,
    required this.onDecision,
  });

  @override
  State<TinderPattern> createState() => _TinderPatternState();
}

class _TinderPatternState extends State<TinderPattern> {
  int currentIndex = 0;
  Offset dragOffset = Offset.zero;
  List<bool?> decisions = [];

  @override
  void initState() {
    super.initState();
    decisions = List.filled(widget.items.length, null);
  }

  void _onSwipe(bool approved) {
    decisions[currentIndex] = approved;
    widget.onDecision(currentIndex, approved);
    setState(() {
      dragOffset = Offset.zero;
      if (currentIndex < widget.items.length - 1) {
        currentIndex++;
      }
    });
  }

  void _undo() {
    if (currentIndex > 0) {
      setState(() {
        currentIndex--;
        decisions[currentIndex] = null;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    if (currentIndex >= widget.items.length) {
      return _buildSummary();
    }

    final item = widget.items[currentIndex];
    final progress = '${currentIndex + 1} of ${widget.items.length}';

    return Column(
      children: [
        // Progress
        Padding(
          padding: const EdgeInsets.all(16),
          child: Text(progress,
            style: TextStyle(color: Colors.white.withValues(alpha: 0.4))),
        ),
        // Card
        Expanded(
          child: GestureDetector(
            onPanUpdate: (details) {
              setState(() { dragOffset += details.delta; });
            },
            onPanEnd: (details) {
              if (dragOffset.dx > 100) {
                _onSwipe(true);  // Right = approve
              } else if (dragOffset.dx < -100) {
                _onSwipe(false); // Left = reject
              } else {
                setState(() { dragOffset = Offset.zero; });
              }
            },
            child: Transform.translate(
              offset: dragOffset,
              child: Transform.rotate(
                angle: dragOffset.dx * 0.001,
                child: _buildCard(item),
              ),
            ),
          ),
        ),
        // Controls
        Padding(
          padding: const EdgeInsets.all(24),
          child: Row(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              IconButton(
                icon: const Icon(Icons.close, color: Colors.red, size: 36),
                onPressed: () => _onSwipe(false),
              ),
              const SizedBox(width: 16),
              if (currentIndex > 0)
                IconButton(
                  icon: Icon(Icons.undo, color: Colors.white.withValues(alpha: 0.4)),
                  onPressed: _undo,
                ),
              const SizedBox(width: 16),
              IconButton(
                icon: const Icon(Icons.check, color: Colors.green, size: 36),
                onPressed: () => _onSwipe(true),
              ),
            ],
          ),
        ),
      ],
    );
  }

  Widget _buildCard(Map<String, dynamic> item) {
    return Container(
      margin: const EdgeInsets.symmetric(horizontal: 48),
      padding: const EdgeInsets.all(32),
      decoration: BoxDecoration(
        color: const Color(0xFF12121A),
        borderRadius: BorderRadius.circular(16),
        border: Border.all(color: Colors.white.withValues(alpha: 0.08)),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(item['title'] ?? '',
            style: const TextStyle(fontSize: 20, fontWeight: FontWeight.w500)),
          const SizedBox(height: 8),
          Text(item['subtitle'] ?? '',
            style: TextStyle(fontSize: 14, color: Colors.white.withValues(alpha: 0.6))),
          const SizedBox(height: 24),
          Expanded(
            child: SingleChildScrollView(
              child: Text(item['content'] ?? '',

            ),
          ),
        ],
      ),
    );

    final rejected = decisions.where((d) => d == false).length;
    return Center(

    );
  }
```

```dart
        mainAxisSize: MainAxisSize.min,
        children: [
          Text(node.title,
            style: const TextStyle(fontWeight: FontWeight.w500, fontSize: 14)),
          if (node.body.isNotEmpty) ...[
            const SizedBox(height: 8),
            Text(node.body,
              style: TextStyle(fontSize: 13, color: Colors.white.withValues(alpha: 0.6)),

class WhiteboardNode {
  final String id;
  final String title;
  final String body;
  double x;
  double y;
  final List<String> connections; // IDs of connected nodes

  WhiteboardNode({
    required this.id,
    required this.title,
    this.body = '',
    this.x = 0,
    this.y = 0,
    this.connections = const [],
  });
}

class _ConnectionPainter extends CustomPainter {
  final List<WhiteboardNode> nodes;
  _ConnectionPainter(this.nodes);

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.white.withValues(alpha: 0.1)
      ..strokeWidth = 1.5
      ..style = PaintingStyle.stroke;

    final nodeMap = {for (var n in nodes) n.id: n};
    for (final node in nodes) {
      for (final targetId in node.connections) {
        final target = nodeMap[targetId];
        if (target != null) {
          canvas.drawLine(
            Offset(node.x + 100, node.y + 30),
            Offset(target.x + 100, target.y + 30),
            paint,
          );
        }
      }
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;
}
```

```python
# agent/auth.py
import hashlib
import secrets
import sqlite3

class UserExists(Exception):
    pass

def _get_db(path: str) -> sqlite3.Connection:
    conn = sqlite3.connect(path)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS users (
            username TEXT PRIMARY KEY,
            password_hash TEXT NOT NULL
        )
    """)
    conn.commit()
    return conn

def _hash(password: str, salt: str) -> str:
    return hashlib.pbkdf2_hmac(
        "sha256", password.encode(), salt.encode(), 100_000
    ).hex()

def create_user(db_path: str, username: str, password: str):
    db = _get_db(db_path)
    existing = db.execute(
        "SELECT 1 FROM users WHERE username = ?", (username,)
    ).fetchone()
    if existing:
        raise UserExists(f"User '{username}' already exists")
    salt = secrets.token_hex(16)
    password_hash = f"{salt}:{_hash(password, salt)}"
    db.execute(
        "INSERT INTO users (username, password_hash) VALUES (?, ?)",
        (username, password_hash),
    )
    db.commit()

def authenticate(db_path: str, username: str, password: str) -> bool:
    db = _get_db(db_path)
    row = db.execute(
        "SELECT password_hash FROM users WHERE username = ?", (username,)
    ).fetchone()
    if not row:
        return False
    salt, stored_hash = row[0].split(":")
    return _hash(password, salt) == stored_hash
```
